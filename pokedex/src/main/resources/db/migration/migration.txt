CREATE TABLE Pokemon (
    id_pok integer Primary Key,
    nome varchar(20) NOT NULL,
    tipo varchar(8),
    tamanho decimal(5,2),
    peso decimal(5,2),
    bioma varchar(20)
);

CREATE TABLE Treinador (
    id_tre integer Primary Key,
    nome varchar(50),
    cidade varchar(50)
);

CREATE TABLE Pokemon_Treinador (
    id_pok integer,
    id_tre integer,
    Primary Key(id_pok, id_tre),
    Foreign Key (id_pok) REFERENCES Pokemon(id_pok),
    Foreign Key (id_tre) REFERENCES Treinador(id_tre)
);

CREATE TABLE Ataques (
    id_ataq integer Primary Key,
    nome varchar(50) NOT NULL,
    tipo varchar(20),
    dano integer,
    precisao integer
);

CREATE TABLE Pokemon_Ataque (
    id_pok integer,
    id_ataq integer,
    Primary Key(id_pok, id_ataq),
    Foreign Key (id_pok) REFERENCES Pokemon(id_pok),
    Foreign Key (id_ataq) REFERENCES Ataques(id_ataq)
);

CREATE TABLE Regiao (
    id_reg integer Primary Key,
    nome varchar(50) NOT NULL
);

CREATE TABLE Pokemon_Regiao (
    id_pok integer,
    id_reg integer,
    Primary Key(id_pok, id_reg),
    Foreign Key (id_pok) REFERENCES Pokemon(id_pok),
    Foreign Key (id_reg) REFERENCES Regiao(id_reg)
);
